cmake_minimum_required(VERSION 3.16)
project(stepper_motion_controller VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
add_library(motion_core src/motion_controller.c src/motion_profile.c)
target_include_directories(motion_core PUBLIC include)
if(MSVC)
  target_compile_options(motion_core PRIVATE /W4)
else()
  target_compile_options(motion_core PRIVATE -Wall -Wextra -Wpedantic)
  if(ENABLE_WARNINGS_AS_ERRORS)
    target_compile_options(motion_core PRIVATE -Werror)
  endif()
endif()
add_executable(motion_demo simulator/main.c simulator/virtual_plant.c)
target_include_directories(motion_demo PRIVATE simulator)
target_link_libraries(motion_demo PRIVATE motion_core)
enable_testing()
add_executable(test_motion tests/test_motion.c)
target_link_libraries(test_motion PRIVATE motion_core)
add_test(NAME motion_unit_tests COMMAND test_motion)
